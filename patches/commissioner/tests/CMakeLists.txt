cmake_minimum_required(VERSION 3.16)
project(tnt_legacy_wakeup_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT TNT_COMMISSIONER_SOURCE)
  message(FATAL_ERROR "Provide the generated patched Commissioner source")
endif()
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_LIST_DIR}/../../../tools/prepare_commissioner_source.py"
  --verify-only --output "${TNT_COMMISSIONER_SOURCE}"
  RESULT_VARIABLE source_verification OUTPUT_QUIET)
if(NOT source_verification EQUAL 0)
  message(FATAL_ERROR "Generated Commissioner source verification failed")
endif()
set(OT_COMM_APP OFF CACHE BOOL "" FORCE)
set(OT_COMM_CCM OFF CACHE BOOL "" FORCE)
set(OT_COMM_TEST OFF CACHE BOOL "" FORCE)
set(OT_COMM_JAVA_BINDING OFF CACHE BOOL "" FORCE)
set(CMAKE_EXPORT_NO_PACKAGE_REGISTRY ON)
set(FMT_DEBUG_POSTFIX "" CACHE STRING "" FORCE)
# MbedTLS imports build scripts from its source directory. Keep it immutable.
set(ENV{PYTHONDONTWRITEBYTECODE} 1)
add_subdirectory("${TNT_COMMISSIONER_SOURCE}" commissioner EXCLUDE_FROM_ALL)
if(APPLE)
  target_compile_options(mbedx509 PRIVATE -Wno-error=uninitialized-const-pointer)
endif()
add_executable(legacy-wakeup-tests test_legacy_wakeup.cpp)
target_include_directories(legacy-wakeup-tests PRIVATE "${TNT_COMMISSIONER_SOURCE}/src")
target_link_libraries(legacy-wakeup-tests PRIVATE commissioner fmt::fmt event_core mbedtls)
enable_testing()
add_test(NAME legacy-wakeup-native COMMAND legacy-wakeup-tests)
